```dart
  /// Create a new CmsClient (workspace) and an admin User for the caller in one transaction.
  /// Used by the manage app's setup wizard for first-time users.
  Future<CmsClient> createClientWithOwner(
    Session session, {
    required String clientName,
    required String clientSlug,
  }) async {
    final authInfo = session.authenticated;
    if (authInfo == null) {
      throw Exception('User must be authenticated');
    }

    // Guard: caller already has a workspace
    final existingUser = await User.db.findFirstRow(
      session,
      where: (t) => t.serverpodUserId.equals(authInfo.userIdentifier),
    );
    if (existingUser != null) {
      throw Exception('Account already has a workspace');
    }

    // Validate slug format
    if (!_slugRegex.hasMatch(clientSlug)) {
      throw Exception(
        'Invalid slug: must be 3-63 characters, lowercase alphanumeric and hyphens, '
        'cannot start or end with a hyphen',
      );
    }
    if (_reservedSlugs.contains(clientSlug)) {
      throw Exception('Slug "$clientSlug" is reserved and cannot be used');
    }

    // Check slug uniqueness in CmsClient table
    final existing = await CmsClient.db.findFirstRow(
      session,
      where: (t) => t.slug.equals(clientSlug),
    );
    if (existing != null) {
      throw Exception('Slug "$clientSlug" is already taken');
    }

    // Get user profile for email
    String? email;
    String? userName;
    try {
      final profileRows = await session.db.unsafeQuery(
        r'SELECT "email", "fullName" FROM "serverpod_auth_core_profile" '
        r'WHERE "authUserId" = $1 LIMIT 1',
        parameters: QueryParameters.positional([authInfo.userIdentifier]),
      );
      if (profileRows.isNotEmpty) {
        email = profileRows.first[0] as String?;
        userName = profileRows.first[1] as String?;
      }
    } catch (e) {
      // Profile lookup failed; use identifier as fallback
    }

    return session.db.transaction((transaction) async {
      final client = await CmsClient.db.insertRow(
        session,
        CmsClient(
          name: clientName,
          slug: clientSlug,
          isActive: true,
          createdAt: DateTime.now(),
          updatedAt: DateTime.now(),
        ),
      );

      await User.db.insertRow(
        session,
        User(
          clientId: client.id!,
          email: email ?? authInfo.userIdentifier,
          name: userName,
          role: 'admin',
          isActive: true,
          serverpodUserId: authInfo.userIdentifier,
          createdAt: DateTime.now(),
          updatedAt: DateTime.now(),
        ),
      );

      return client;
    });
  }
```

class _SetupWizardScreenState extends State<SetupWizardScreen> {
  final _clientFormKey = GlobalKey<ShadFormState>();
  final _projectFormKey = GlobalKey<ShadFormState>();
  final _clientNameController = TextEditingController();
  final _projectNameController = TextEditingController();

  int _step = 1;
  bool _isLoading = false;
  String? _error;

  String _generateSlug(String name) {
    return name
        .toLowerCase()
        .replaceAll(RegExp(r'[^a-z0-9]+'), '-')
        .replaceAll(RegExp(r'^-+|-+$'), '');
  }

  Future<void> _createClient() async {
    if (!(_clientFormKey.currentState?.validate() ?? false)) return;

    final name = _clientNameController.text.trim();
    final slug = _generateSlug(name);

    if (slug.length < 3) {
      setState(
          () => _error = 'Workspace name is too short to generate a valid slug.');
      return;
    }

    setState(() {
      _isLoading = true;
      _error = null;
    });

    try {
      final appVM = GetIt.I<AppViewModel>();
      await appVM.client.project.createClientWithOwner(
        clientName: name,
        clientSlug: slug,
      );
      if (mounted) {
        setState(() => _step = 2);
      }
    } catch (e) {
      if (mounted) {
        final message = e.toString();
        if (message.contains('already taken')) {
          setState(() =>
              _error = 'Slug "$slug" is already taken. Try a different name.');
        } else if (message.contains('reserved')) {
          setState(() =>
              _error = 'This name uses a reserved word. Try a different name.');
        } else if (message.contains('already has a workspace')) {
          setState(() => _error = 'Your account already has a workspace.');
        } else {
          setState(
              () => _error = 'Failed to create workspace. Please try again.');
        }
      }
    } finally {
      if (mounted) {
        setState(() => _isLoading = false);
      }
    }
  }

  Future<void> _createProject() async {
    if (!(_projectFormKey.currentState?.validate() ?? false)) return;

    final name = _projectNameController.text.trim();
    final slug = _generateSlug(name);

    if (slug.length < 3) {
      setState(
          () => _error = 'Project name is too short to generate a valid slug.');
      return;
    }

    setState(() {
      _isLoading = true;
      _error = null;
    });

    try {
      final appVM = GetIt.I<AppViewModel>();
      final project = await appVM.client.project.createProject(name, slug);
      if (mounted) {
        appVM.userClients.reload();
        appVM.initClientContext(project.slug);
        context.router.root.navigate(ManageShellRoute(
            clientSlug: project.slug, children: [const OverviewRoute()]));
      }
    } catch (e) {
      if (mounted) {
        final message = e.toString();
        if (message.contains('already taken')) {
          setState(() =>
              _error = 'Slug "$slug" is already taken. Try a different name.');
        } else {
          setState(
              () => _error = 'Failed to create project. Please try again.');
        }
      }
    } finally {
      if (mounted) {
        setState(() => _isLoading = false);
      }
    }
  }
